41template <typename TDomain, typename TAlgebra>
42class SortedGridFunctionOrdering
43{
44 typedef GridFunction<TDomain, TAlgebra> TGridFunction;
45 typedef GridFunctionNumberData<GridFunction<TDomain, TAlgebra> > TGridFunctionNumberData;
46
47 typedef std::vector<size_t> ordering_container_type;
48 typedef IOrderingAlgorithm<TAlgebra, ordering_container_type> ordering_algo_type;
49
50 typedef typename TGridFunction::template traits<Vertex>::const_iterator VertexConstIterator;
51
52public:
53 SortedGridFunctionOrdering(SmartPtr<TGridFunction> spGridFct, SmartPtr<ordering_algo_type> spOrdAlgo, const char* name)
54 {
55 m_u = spGridFct->clone_without_values();
56 m_name = name;
57 ordering_container_type &o = spOrdAlgo->ordering();
58
59 std::vector<DoFIndex> ind(1);
60 size_t k = 0;
61 for(VertexConstIterator iter = m_u->template begin<Vertex>(); iter != m_u->template end<Vertex>(); ++iter)
62 {
63 // get vertex
64 Vertex* vrt = *iter;
65
66 // get vector holding all indices on the vertex
67 m_u->inner_dof_indices(vrt, 0, ind);
68 DoFRef(*m_u, ind[0]) = o[k++];
69 }
70 }
71
72 SmartPtr<TGridFunctionNumberData> get(){
73 return SmartPtr<TGridFunctionNumberData>(new TGridFunctionNumberData(m_u, m_name));
74 }
75
76private:
77 SmartPtr<TGridFunction> m_u;
78 const char* m_name;
79};
