108 void remove (SmartPtr<p_process_type> p)
109 {
110 size_t i = 0;
111 while (i < m_pp_chain.size ())
112 {
113 if (m_pp_chain[i].get () == p.get ())
114 m_pp_chain.erase (m_pp_chain.begin () + i);
115 else
116 i++;
117 }
118 }

120private:
