43template<typename TBlock>
44bool BlockVectorFiniteAndNotTooBig(TBlock &v, double tooBigValue=1e24)
45{
46 for(size_t j=0; j< GetSize(v); j++)
47 {
48 double d = BlockRef(v, j);
49 if(d > tooBigValue || d < -tooBigValue || std::isfinite(d) == false)
50 return false;
51 }
52 return true;
53}
54
55template<typename TBlock>
56bool BlockMatrixFiniteAndNotTooBig(TBlock &m, double tooBigValue=1e24)
57{
58 for(size_t r=0; r<GetRows(m); r++)
59 for(size_t c=0; c< GetCols(m); c++)
60 {
61 double d = BlockRef(m, r, c);
62 if(d > tooBigValue || d < -tooBigValue || std::isfinite(d) == false)
63 return false;
64 }
65
66 return true;
67}

70template<typename TVector>
71bool IsFiniteAndNotTooBig(const TVector &v, double tooBigValue=1e24)
72{
73 for(size_t i=0; i<v.size(); i++)
74 if(BlockVectorFiniteAndNotTooBig(v[i], tooBigValue) == false)
75 return false;
76 return true;
77}
