40class RotatingCone2d
41 : public StdGlobPosData<RotatingCone2d, number, 2>
42{
43 public:
44 RotatingCone2d(double _eps, double _cx, double _cy, double _ax, double _ay,
45 double _nu, double _delta)
46 : eps(_eps), cx(_cx), cy(_cy), ax(_ax), ay(_ay), nu(_nu), delta(_delta)
47 {};
48
49 inline void evaluate(number& val, const MathVector<2>& pos, number time, int si) const
50 {
51 const number t = time;
52 const number x = pos[0];
53 const number y = pos[1];
54
55 const number xRot = cos(nu*t) * (x-cx) - sin(nu*t) * (y-cy);
56 const number yRot = sin(nu*t) * (x-cx) + cos(nu*t) * (y-cy);
57
58 const number expo = -((xRot - ax)*(xRot - ax) + (yRot - ay)*(yRot - ay)) / (delta + 4*eps*t);
59 const number scale = delta/(delta+4*eps*t);
60
61 val = scale * exp(expo);
62 }
63
64 protected:
65 double eps, cx, cy, ax, ay, nu, delta;
66};
67
68}
