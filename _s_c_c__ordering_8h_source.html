89 void compute(){
90 unsigned n = boost::num_vertices(g);
91
92 if(n == 0){
93 UG_THROW(name() << "::compute: Graph is empty!");
94 return;
95 }
96
97 std::vector<int> component(n), discover_time(n);
98 std::vector<boost::default_color_type> color(n);
99 std::vector<vd_t> root(n);
100 size_t num_components = boost::strong_components(g,
101 boost::make_iterator_property_map(component.begin(), boost::get(boost::vertex_index, g)),
102 boost::root_map(boost::make_iterator_property_map(root.begin(), boost::get(boost::vertex_index, g)))
103 .color_map(
104 boost::make_iterator_property_map(color.begin(), boost::get(boost::vertex_index, g)))
105 .discover_time_map(boost::make_iterator_property_map(
106 discover_time.begin(), boost::get(boost::vertex_index, g))));
107
108
109 std::vector<std::vector<vd_t> > comp_members(num_components);
110 for(unsigned i = 0; i < n; ++i){
111 comp_members[component[i]].push_back(i);
112 }
113
114 //create scc meta graph
115 scc_g = G_t(num_components);
116
117 adj_iter nIt, nEnd;
118 size_t i_comp, n_comp;
119 for(unsigned i = 0; i < n; ++i){
120 i_comp = component[i];
121 for(boost::tie(nIt, nEnd) = boost::adjacent_vertices(i, g); nIt != nEnd; ++nIt){
122 n_comp = component[*nIt];
123 if(i_comp != n_comp && !boost::edge(i_comp, n_comp, scc_g).second){
124 boost::add_edge(i_comp, n_comp, scc_g);
125 }
126 }
127 }
128
129 O_t scc_topo_ordering(num_components);
130 topological_ordering_core_directed(scc_topo_ordering, scc_g, true); //true = inverse
131
132 //scc_topo_ordering is now a topological ordering of scc_g
133
134 o.resize(n);
135
136 if(m_spOrderingSubAlgo.invalid()){
137 UG_LOG(name() << "::compute: not using ordering subalgo\n");
138 //enumerate members of components where components are iterated using the
139 //topological ordering of the scc meta graph
140
141 size_t k = 0;
142 for(unsigned i = 0; i < num_components; ++i){
143 for(unsigned j = 0; j < comp_members[scc_topo_ordering[i]].size(); ++j){
144 o[comp_members[scc_topo_ordering[i]][j]] = k++;
145 }
146 }
147 }
148 else{
149 UG_LOG(name() << ":compute: using " << m_spOrderingSubAlgo->name() << " as subalgo\n");
150
151 size_t k = 0;
152 for(unsigned i = 0; i < num_components; ++i){
153 size_t c_size = comp_members[scc_topo_ordering[i]].size();
154
155 if(c_size == 1){
156 o[comp_members[scc_topo_ordering[i]][0]] = k++;
157 }
158 else{
159 m_spOrderingSubAlgo->init(m, comp_members[scc_topo_ordering[i]]);
160 m_spOrderingSubAlgo->compute();
161 std::vector<size_t>& sub_o = m_spOrderingSubAlgo->ordering();
162 std::vector<size_t> inv_sub_o;
163 GetInversePermutation(sub_o, inv_sub_o);
164
165 for(unsigned j = 0; j < c_size; ++j){
166 o[comp_members[scc_topo_ordering[i]][inv_sub_o[j]]] = k++;
167 }
168 }
169 }
170
171 UG_COND_THROW(k != n, "k!=n, k=" << k << ", n=" << n);
172 }
173
174 //reset
175 scc_g = G_t(0);
176 g = G_t(0);
177
178 #ifdef UG_DEBUG
179 check();
180 #endif
181 }

195 void init(M_t* A){
196 //TODO: replace this by UG_DLOG if permutation_util does not depend on this file anymore
197 #ifdef UG_ENABLE_DEBUG_LOGS

199 #endif
200

202
203 unsigned rows = A->num_rows();
204
205 g = G_t(rows);
206
207 for(unsigned i = 0; i < rows; i++){
208 for(typename M_t::row_iterator conn = A->begin_row(i); conn != A->end_row(i); ++conn){
209 if(conn.value() != 0.0 && conn.index() != i){
210 boost::add_edge(i, conn.index(), g);
211 }
212 }
213 }
