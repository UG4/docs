46{
47 public:
48 IsInSubset(const ISubsetHandler& sh, int subsetIndex) :
49 m_sh(sh),
50 m_si(subsetIndex) {}
51
52 bool operator() (Vertex* v) const {return callback(v);}
53 bool operator() (Edge* e) const {return callback(e);}
54 bool operator() (Face* f) const {return callback(f);}
55 bool operator() (Volume* v) const {return callback(v);}
