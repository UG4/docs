61template <class TLayout>
62void LogLayoutStructure(TLayout& layout, const char* prefix = "")
63{
64 using namespace std;
65
66 typedef typename TLayout::Interface Interface;
67 typedef typename TLayout::iterator InterfaceIter;
68
69 PCLLOG(prefix << "-- PCL_DEBUG: layout-structure --\n");
70 PCLLOG(prefix << "---- num_levels: " << layout.num_levels() << endl);
71
72 for(size_t lvl = 0; lvl < layout.num_levels(); ++lvl)
73 {
74 PCLLOG(prefix << "---- interfaces on level " << lvl << " (proc id, size): ");
75
76 for(InterfaceIter iiter = layout.begin(lvl);
77 iiter != layout.end(lvl); ++iiter)
78 {
79 Interface& interface = layout.interface(iiter);
80 PCLLOG("(" << layout.proc_id(iiter) << ", " << interface.size() << "), ");
81 }
82 PCLLOG(endl);
83 }
84}

92template <class TType, class TLayoutMap>

101
102 for(iterator iter = lm.template layouts_begin<TType>();
103 iter != lm.template layouts_end<TType>(); ++iter)
104 {
105 // get the key
106 PCLLOG("---- has key: " << iter->first << endl);
107
108 // log the layout structure
109 Layout& layout = iter->second;
110 LogLayoutStructure(layout, "----");
111 }
