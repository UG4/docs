105template<class TGeomObj>
106typename geometry_traits<TGeomObj>::iterator
107Grid::create_and_replace(typename geometry_traits<TGeomObj>::grid_base_object* pReplaceMe)
108{
109 STATIC_ASSERT(geometry_traits<TGeomObj>::CONTAINER_SECTION != -1
110 && geometry_traits<TGeomObj>::BASE_OBJECT_ID != -1,
111 invalid_geometry_type);
112
113 TGeomObj* geomObj = new TGeomObj;
114
115 if(geomObj->reference_object_id() == pReplaceMe->reference_object_id())
116 {
117 register_and_replace_element(geomObj, pReplaceMe);
118 return iterator_cast<typename geometry_traits<TGeomObj>::iterator>(get_iterator(geomObj));
119 }
120 else
121 {
122 LOG("ERROR in Grid::create_and_replace(...): reference objects do not match!");
123 assert(!"ERROR in Grid::create_and_replace(...): reference objects do not match!");
124 delete geomObj;
125 return end<TGeomObj>();
126 }

140// erase
141template <class GeomObjIter>
142void Grid::erase(const GeomObjIter& iterBegin, const GeomObjIter& iterEnd)
143{
144 GeomObjIter iter = iterBegin;
145 while(iter != iterEnd)
146 {
147 GeomObjIter tmpIter = iter;
148 ++iter;
149 erase(*tmpIter);
150 }
151}

153template <class TGeomObj>
154void Grid::clear()
155{
156 while(begin<TGeomObj>() != end<TGeomObj>())
157 erase(*begin<TGeomObj>());
158}

229template <class TGeomObj>
230size_t Grid::num() const
231{

233 invalid_GeomObj);
234

239
240 return element_storage<TGeomObj>().m_sectionContainer.num_elements(secIndex);

542// neighbourhood access
543template <class TGeomObj>
544Edge* Grid::find_edge_in_associated_edges(TGeomObj* obj, const EdgeVertices& ev)
545{
546 GRID_PROFILE_FUNC();
547
548 AssociatedEdgeIterator iterEnd = associated_edges_end(obj);
549 for(AssociatedEdgeIterator iter = associated_edges_begin(obj);
550 iter != iterEnd; ++iter)
551 {
552 Edge* e = *iter;
553 if(CompareVertices(e, &ev))
554 return e;
555 }
556 return NULL;
557}
558
559
560template <class TGeomObj>
561Face* Grid::find_face_in_associated_faces(TGeomObj* obj, const FaceVertices& fv)
562{
563 GRID_PROFILE_FUNC();
564
565 unsigned long key = hash_key(&fv);
566 AssociatedFaceIterator iterEnd = associated_faces_end(obj);
567 for(AssociatedFaceIterator iter = associated_faces_begin(obj);
568 iter != iterEnd; ++iter)
569 {
570 Face* f = *iter;
571 if(key == hash_key(f))
572 {
573 if(CompareVertices(f, &fv))
574 return f;
575 }
576 }
577
578 return NULL;
579}
580
581template <class TGeomObj>
582Volume* Grid::find_volume_in_associated_volumes(TGeomObj* obj,
583 const VolumeVertices& vv)
584{
585 GRID_PROFILE_FUNC();
586
587 unsigned long key = hash_key(&vv);
588 AssociatedVolumeIterator iterEnd = associated_volumes_end(obj);
589 for(AssociatedVolumeIterator iter = associated_volumes_begin(obj);
590 iter != iterEnd; ++iter)
591 {
592 Volume* v = *iter;
593 if(key == hash_key(v))
594 {
595 if(CompareVertices(v, &vv))
596 return v;
597 }
598 }
599
600 return NULL;
601}
