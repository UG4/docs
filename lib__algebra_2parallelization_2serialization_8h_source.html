183template<class TOStream>
184void Serialize(TOStream &buf, const pcl::ProcessCommunicator &ic)
185{
186 int s;
187 if(ic.is_world())
188 s = -1;
189 else
190 s = ic.size();
191 Serialize(buf, s);
192
193 if(s > 0)
194 {
195 for(int i=0; i<s; i++)
196 Serialize(buf, ic.get_proc_id(i));
197 }
198}

201template<class TIStream>
202void Deserialize(TIStream &buf, pcl::ProcessCommunicator &ic)
203{
204 int s = Deserialize<int>(buf);
205
206 if(s == -1)
207 {
208 if(ic.is_world()) return;
209 ic = pcl::ProcessCommunicator(pcl::PCD_WORLD);
210 }
211 else if (s == 0)
212 {
213 if(ic.empty()) return;
214 ic = pcl::ProcessCommunicator(pcl::PCD_EMPTY);
215 }
216 else
217 {
218 std::vector<int> procs;
219 for(int i=0; i<s; i++)
220 procs.push_back(Deserialize<int>(buf));
221 ic = pcl::ProcessCommunicator::create_communicator(procs);
222 }
223}
