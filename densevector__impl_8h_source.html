53template<typename TStorage>
54DenseVector<TStorage> &
55DenseVector<TStorage>::operator = (const DenseVector<TStorage> &rhs)
56{
57 if(this == &rhs) return *this;
58 if(size() != rhs.size()) resize(rhs.size());
59
60 for(size_type i=0; i<size(); i++)
61 entry(i) = rhs[i];
62 return *this;
63}
