59template <typename TDomain, typename TAlgebra>
60class BiotProjection
61{
62public:
63 typedef BiotProblem<TDomain, TAlgebra> TBiotProblem;
64 typedef typename TBiotProblem::TDomainDisc TDomainDisc;
65
66 using TSolver = ILinearOperatorInverse<typename TAlgebra::vector_type>;
67 using TGridFunction = typename TBiotProblem::TGridFunction;
68protected:
69 using TTimeIntegrator = ConstStepLinearTimeIntegrator<TDomain,TAlgebra>;
70
71public:
73 BiotProjection(SmartPtr<TDomainDisc> dd0, SmartPtr<TSolver> solver)
74 : m_solver(solver), m_dd0(dd0), time_integrator(SPNULL) {}
75
77 void apply (SmartPtr<TGridFunction> sol)
78 {
79 // Create time integrator (if required)
80 if (time_integrator.invalid())
81 {
82 using TTimeDisc = ThetaTimeStep<TAlgebra>;
83 SmartPtr<TTimeDisc> time_disc = make_sp<TTimeDisc> (new TTimeDisc(m_dd0));
84 time_integrator = make_sp<TTimeIntegrator> (new TTimeIntegrator(time_disc, m_solver));
85 }
86
87 // Perform one step.
88 double tau0 = 1.0; // dummy tau
89 time_integrator->set_time_step(tau0);
90 time_integrator->set_precision_bound(1e-12);
91 time_integrator->apply(sol, tau0, sol, 0.0);
92
93 }
94
95protected:
96
97 SmartPtr<TSolver> m_solver;
98 SmartPtr<TDomainDisc> m_dd0; // domain disc for initial values
99 SmartPtr<TTimeIntegrator> time_integrator;
100};
