147 void set_density(SmartPtr<CplUserData<number,dim> > data)
148 {
149 // remove old data
150 SmartPtr<ICplUserData<dim> > oldData = m_imDensityCo.data();
151 if (oldData.valid())
152 {
153 m_exFractDarcyVel->remove_needed_data(oldData);
154 m_exOrthoFractDarcyVel->remove_needed_data(oldData);
155 }
156 oldData = m_imDensityIP.data();
157 if (oldData.valid())
158 {
159 m_exFractDarcyVel->remove_needed_data(oldData);
160 m_exOrthoFractDarcyVel->remove_needed_data(oldData);
161 }
162
163 // connect to import
164 m_imDensityCo.set_data(data);
165 m_imDensityIP.set_data(data);
166 if (m_spVolStabData.valid())
167 m_imOldDensityCo.set_data(data);
168 else
169 m_imOldDensityCo.set_data(SPNULL);
170
171 // reset the data in the old density
172 set_volume_stabilization(m_spVolStabData);
173
174 // darcy velocity depends on density
175 m_exFractDarcyVel->add_needed_data(data);
176 m_exOrthoFractDarcyVel->add_needed_data(data);
177 }
