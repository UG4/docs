42template <typename TDomain, typename TAlgebra>
43NavierStokesInflowFV<TDomain,TAlgebra>::
44NavierStokesInflowFV(SmartPtr< NavierStokesFV<TDomain> > spMaster)
45 : m_spNeumannDisc(new NeumannBoundaryFV<TDomain>(spMaster->symb_fcts()[dim].c_str())),
46 m_spDirichletConstraint(new DirichletBoundary<TDomain,TAlgebra>),
47 m_spMaster(spMaster)
48{
49 const std::vector<std::string>& vFctName = m_spMaster->symb_fcts();
50
51 if(vFctName.size() != TDomain::dim + 1)
52 UG_THROW("NavierStokesInflow::set_functions: This Boundary "
53 "Condition works on exactly dim+1 (velocity+pressure) "
54 "components, but "<<vFctName.size()<<"components given.");
55}

57template <typename TDomain, typename TAlgebra>
58void NavierStokesInflowFV<TDomain,TAlgebra>::
59add(SmartPtr<CplUserData<MathVector<dim>, dim> > user, const char* subsetsBND)
60{
61 const std::vector<std::string>& vFctName = m_spMaster->symb_fcts();
62 if(vFctName.empty())
63 UG_THROW("NavierStokesInflow::add: Symbolic names for"
64 " velocity and pressure not set. Please set them first.");
65
66 std::string innerSubsets;
67 for(size_t s = 0; s < m_spMaster->symb_subsets().size(); ++s){
68 if(s > 0) innerSubsets.append(",");
69 innerSubsets.append(m_spMaster->symb_subsets()[s]);
70 }
71
72
73 m_spNeumannDisc->add(user, subsetsBND, innerSubsets.c_str());
74
75 std::string velNames;
76 for(int i=0;i<TDomain::dim; ++i)
77 {
78 if(i>0) velNames.append(",");
79 velNames.append(vFctName[i]);
80 }
81 m_spDirichletConstraint->add(user.template cast_dynamic<UserData<MathVector<dim>, dim> >(), velNames.c_str(), subsetsBND);
82}
