74 template <class TVal>
75 TVal get_val(const char* name)
76 {
77 lua_getglobal(m_luaState, name);
78 if(!bridge::lua::LuaParsing<TVal>::check(m_luaState, -1)){
79 lua_pop(m_luaState, 1);
80 UG_THROW("LuaShell error: Couldn't convert " << name << " to requested type.");
81 }
82
83 TVal val = bridge::lua::LuaParsing<TVal>::get(m_luaState, -1);
84 lua_pop(m_luaState, 1);
85 return val;
86 }
