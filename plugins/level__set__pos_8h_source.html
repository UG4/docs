162 bool get_height_at
163 (
164 const MathVector<dim-1> & xy,
165 number& height
166 )
167 {
168 std::vector<number> z;
169 get_height (xy, z);
170 if (z.size () == 0) return false;
171 number max_z = z[0];
172 for (size_t i = 1; i < z.size (); i++)
173 if (z[i] > max_z)
174 max_z = z[i];
175 height = max_z;
176 return true;
177 }
