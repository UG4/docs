50template <>
51inline number cone_mollifier<1>
52(
53 number d,
54 number smooth_len
55)
56{
57 d /= smooth_len;
58 if (d >= 1) return 0;
59 return (1 - d) / (smooth_len / 2);
60}
61
65template <>
66inline number cone_mollifier<2>
67(
68 number d,
69 number smooth_len
70)
71{
72 d /= smooth_len;
73 if (d >= 1) return 0;
74 return (1 - d) / (PI * smooth_len * smooth_len / 3);
75}
