46 public:
47 void set_min(const vector3& val) {min = val;}

49 const vector3& get_min() const {return min;}
50 const vector3& get_max() const {return max;}
51
52 void global_to_local (const vector3& glob, vector3& locOut)
53 {
54 for(size_t i = 0; i < 3; ++i){
55 number d = max[i] - min[i];
56 if(d != 0)
57 locOut[i] = (glob[i] - min[i]) / d;
58 else
59 locOut[i] = 0;
60 }
61 }
62
63 void local_to_global (const vector3& loc, vector3& globOut)
64 {
65 for(size_t i = 0; i < 3; ++i){
66 number d = max[i] - min[i];
67 globOut[i] = loc[i] * d + min[i];
68 }
69 }
70
71 vector3 min;
72 vector3 max;
73};
