53class IScaledHookeLaw : public HookeLaw<TDomain>
54{
55 private:
57 typedef HookeLaw<TDomain> base_type;
58
60 typedef IScaledHookeLaw<TDomain> this_type;
61
62
63 public:
65 static const int dim = base_type::dim;
66
68 typedef typename base_type::TBaseElem TBaseElem;
69
70 public:
72 IScaledHookeLaw( SmartPtr< GridFunction<TDomain, CPUAlgebra> > spScaling,
73 SmartPtr< GridFunction<TDomain, CPUAlgebra> > spEnergy)
74 : HookeLaw<TDomain>(),
75 m_spScaling(spScaling), m_spEnergy(spEnergy),
76 m_pScaling_elem(NULL), m_pEnergy_elem(NULL)
77 {}
78
80 ~IScaledHookeLaw(){};
81
83 void init_internal_vars(TBaseElem* elem, const size_t numIP)
84 {
85 if (!base_type::m_bInit){
86 base_type::m_bInit = true;
87
88 m_spScaling->set(1.0);
89 m_spEnergy->set(0.0);
90 }
91 }
92
93 void internal_vars(TBaseElem* elem)
94 {
95 const size_t fct = 0; // \todo: generalize
96 std::vector<DoFIndex> ind;
97 if(m_spScaling->inner_dof_indices(elem, fct, ind) != 1)
98 UG_THROW("Wrong number dofs");
99
100 m_pScaling_elem = & DoFRef(*m_spScaling, ind[0]);
101 m_pEnergy_elem = &DoFRef(*m_spEnergy, ind[0]);
102 }
103
104 virtual void post_process_energy_on_curr_elem(){}
105
106 public:
107 virtual number scaling_on_curr_elem() {return *m_pScaling_elem;}
108 number& energy_on_curr_elem() {return *m_pEnergy_elem;}
109
110 protected:
111 SmartPtr<GridFunction<TDomain, CPUAlgebra > > m_spScaling;
112 SmartPtr<GridFunction<TDomain, CPUAlgebra > > m_spEnergy;
113
114 number* m_pScaling_elem;
115 number* m_pEnergy_elem;
116};

126class DamageLaw

135

138 using base_type::m_pScaling_elem;
139
140 public:
142 static const int dim = base_type::dim;
143
145 typedef typename base_type::TBaseElem TBaseElem;
146
147 public:
149 DamageLaw( SmartPtr< GridFunction<TDomain, CPUAlgebra> > spF,

190 public:
192 TopologyOptimLaw( SmartPtr< GridFunction<TDomain, CPUAlgebra> > spChi,
193 SmartPtr< GridFunction<TDomain, CPUAlgebra> > spDrivingForce,
194 int expPenalize)
195 : IScaledHookeLaw<TDomain>(spChi, spDrivingForce),
196 m_expPenalize(expPenalize)
197 {}
198
200 ~TopologyOptimLaw() {};
201
202 virtual number scaling_on_curr_elem() {return std::pow(*m_pScaling_elem, m_expPenalize);}
203
204
205 virtual void post_process_energy_on_curr_elem()
206 {
207 (*m_pEnergy_elem) = ((m_expPenalize-1) * (*m_pEnergy_elem)) / (*m_pScaling_elem);
208 }
209
210 protected:
211 int m_expPenalize;
212};
