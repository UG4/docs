158 void set_density(SmartPtr<CplUserData<number,dim> > data)
159 {
160 // remove old data
161 SmartPtr<ICplUserData<dim> > oldData = m_imDensityCo.data();
162 if (oldData.valid())
163 {
164 m_exFractDarcyVel->remove_needed_data(oldData);
165 m_exOrthoFractDarcyVel->remove_needed_data(oldData);
166 }
167 oldData = m_imDensityIP.data();
168 if (oldData.valid())
169 {
170 m_exFractDarcyVel->remove_needed_data(oldData);
171 m_exOrthoFractDarcyVel->remove_needed_data(oldData);
172 }
173
174 // connect to import
175 m_imDensityCo.set_data(data);
176 m_imDensityIP.set_data(data);
177 if (m_spVolStabData.valid())
178 m_imOldDensityCo.set_data(data);
179 else
180 m_imOldDensityCo.set_data(SPNULL);
181
182 // reset the data in the old density
183 set_volume_stabilization(m_spVolStabData);
184
185 // darcy velocity depends on density
186 m_exFractDarcyVel->add_needed_data(data);
187 m_exOrthoFractDarcyVel->add_needed_data(data);
188 }
