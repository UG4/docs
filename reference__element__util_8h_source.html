40inline int ReferenceElementDimension(ReferenceObjectID roid)
41{
42 switch(roid)
43 {
44 case ROID_VERTEX: return 0;
45 case ROID_EDGE: return 1;
46 case ROID_TRIANGLE: return 2;
47 case ROID_QUADRILATERAL: return 2;
48 case ROID_TETRAHEDRON: return 3;
49 case ROID_OCTAHEDRON: return 3;
50 case ROID_PYRAMID: return 3;
51 case ROID_PRISM: return 3;
52 case ROID_HEXAHEDRON: return 3;
53 default:UG_THROW("ReferenceElementDimension: ReferenceObjectId "
54 <<roid<<" not found.");
55 }
56}

63inline MathVector<1> ReferenceElementCenter<1>(ReferenceObjectID roid)
64{
65 switch(roid)

69 }
70}

72template <>
73inline MathVector<2> ReferenceElementCenter<2>(ReferenceObjectID roid)
74{
75 switch(roid)
76 {
77 case ROID_TRIANGLE: return MathVector<2>(1.0/3.0, 1.0/3.0);
78 case ROID_QUADRILATERAL: return MathVector<2>(0.5, 0.5);
79 default: UG_THROW("ReferenceObject "<<roid<<" not found in dim 2.");
80 }
81}
82
83template <>
84inline MathVector<3> ReferenceElementCenter<3>(ReferenceObjectID roid)
85{
86 switch(roid)
87 {
88 case ROID_TETRAHEDRON: return MathVector<3>(0.25, 0.25, 0.25);
89 case ROID_PYRAMID: return MathVector<3>(2./5., 2./5., 1./5.);
90 case ROID_PRISM: return MathVector<3>(2./6., 2./6., 0.5);
91 case ROID_HEXAHEDRON: return MathVector<3>(0.5, 0.5, 0.5);
92 case ROID_OCTAHEDRON: return MathVector<3>(2./6., 2./6., 0.0);
93 default: UG_THROW("ReferenceObject "<<roid<<" not found in dim 3.");
94 }
95}
